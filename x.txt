def classify_result(results):
    if not results:
        return "NO_MATCH", None

    top = results[0]
    score = top["score"]

    # 🔥 Tune these later
    if score >= 0.90:
        return "HIGH_CONFIDENCE", top
    elif score >= 0.80:
        return "MEDIUM_CONFIDENCE", top
    else:
        return "LOW_CONFIDENCE", None

from app.search import search_similar
from app.decision import classify_result


def process_issue(issue):
    results = search_similar(issue)

    decision, best_match = classify_result(results)

    return {
        "decision": decision,
        "best_match": best_match,
        "top_results": results
    }
