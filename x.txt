async function analyzeIssue() {

    const testcase = document.getElementById("testcase").value;
    const error_message = document.getElementById("error_message").value;
    const ims_link = document.getElementById("ims_link").value;

    document.getElementById("loading").classList.remove("hidden");
    document.getElementById("result").classList.add("hidden");

    try {

        const response = await fetch("/analyze_issue", {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify({
                testcase,
                error_message,
                ims_link
            })
        });

        const data = await response.json();

        document.getElementById("decision").innerText = data.decision;
        document.getElementById("matched_issue").innerText = data.matched_issue || "N/A";
        document.getElementById("score").innerText = data.score || "N/A";
        document.getElementById("analysis").innerText = data.analysis || "No analysis found";

        document.getElementById("result").classList.remove("hidden");

    } catch (err) {
        alert("Error calling API");
        console.error(err);
    }

    document.getElementById("loading").classList.add("hidden");
}
