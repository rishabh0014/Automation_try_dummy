import numpy as np
import faiss
import os

EMBEDDINGS_FILE = "models/embeddings.npy"
INDEX_FILE = "models/faiss.index"


def main():
    print("📥 Loading embeddings...")

    embeddings = np.load(EMBEDDINGS_FILE)

    dim = embeddings.shape[1]
    print(f"📐 Embedding dimension: {dim}")

    # Normalize for cosine similarity
    faiss.normalize_L2(embeddings)

    # Create index
    index = faiss.IndexFlatIP(dim)  # Inner Product = cosine after normalize

    print("⚙️ Building FAISS index...")
    index.add(embeddings)

    os.makedirs("models", exist_ok=True)
    faiss.write_index(index, INDEX_FILE)

    print("✅ FAISS index saved!")


if __name__ == "__main__":
    main()



import faiss
import numpy as np
import json

from app.embedder import get_embedding
from app.utils import build_embedding_text

INDEX_FILE = "models/faiss.index"
MAPPING_FILE = "models/mapping.json"

# Load once (important for performance)
index = faiss.read_index(INDEX_FILE)

with open(MAPPING_FILE, "r") as f:
    mapping = json.load(f)


def search_similar(issue, top_k=5):
    text = build_embedding_text(issue)

    query_embedding = get_embedding(text)
    query_embedding = np.array([query_embedding]).astype("float32")

    # Normalize (same as index)
    faiss.normalize_L2(query_embedding)

    scores, indices = index.search(query_embedding, top_k)

    results = []

    for score, idx in zip(scores[0], indices[0]):
        if idx == -1:
            continue

        result = mapping[idx]
        result["score"] = float(score)

        results.append(result)

    return results
