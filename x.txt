import requests

OLLAMA_URL = "http://localhost:11434/api/embeddings"
MODEL_NAME = "nomic-embed-text"  # change if needed


def get_embedding(text: str):
    payload = {
        "model": MODEL_NAME,
        "prompt": text
    }

    response = requests.post(OLLAMA_URL, json=payload)

    if response.status_code != 200:
        raise Exception(f"Ollama error: {response.text}")

    return response.json()["embedding"]



def build_embedding_text(issue, description=None):
    parts = []

    if issue.get("testcase"):
        parts.append(f"Testcase: {issue['testcase']}")

    if issue.get("error_message"):
        parts.append(f"Error: {issue['error_message']}")

    if description:
        parts.append(f"Description: {description}")

    return "\n".join(parts)




import json
import numpy as np
import os

from app.embedder import get_embedding
from app.utils import build_embedding_text

INPUT_FILE = "data/processed/issues.json"
OUTPUT_FILE = "models/embeddings.npy"
MAPPING_FILE = "models/mapping.json"


def main():
    print("📥 Loading issues...")

    with open(INPUT_FILE, "r") as f:
        issues = json.load(f)

    embeddings = []
    mapping = []

    for idx, issue in enumerate(issues):
        try:
            text = build_embedding_text(issue)

            emb = get_embedding(text)

            embeddings.append(emb)
            mapping.append(issue)

            if idx % 50 == 0:
                print(f"Processed {idx} issues...")

        except Exception as e:
            print(f"⚠️ Skipping issue {issue.get('issue_id')}: {e}")

    embeddings = np.array(embeddings)

    os.makedirs("models", exist_ok=True)

    np.save(OUTPUT_FILE, embeddings)

    with open(MAPPING_FILE, "w") as f:
        json.dump(mapping, f)

    print("✅ Embeddings saved!")


if __name__ == "__main__":
    main()
