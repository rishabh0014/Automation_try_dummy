from app.engine import process_issue

def get_user_input():
    print("\nEnter Issue Details (type 'exit()' to quit)\n")

    testcase = input("Testcase: ")
    if testcase.strip() == "exit()":
        return None

    error_message = input("Error Message: ")
    if error_message.strip() == "exit()":
        return None

    ims_link = input("IMS Link: ")
    if ims_link.strip() == "exit()":
        return None

    return {
        "testcase": testcase,
        "error_message": error_message,
        "ims_link": ims_link
    }


def main():
    print("🚀 PCIeJ Similarity CLI Tool Started")

    while True:
        issue = get_user_input()

        if issue is None:
            print("👋 Exiting...")
            break

        print("\n🔍 Processing...\n")

        result = process_issue(issue, auto_update=True)

        print("📊 Decision:", result["decision"])

        if result["best_match"]:
            best = result["best_match"]

            print("\n✅ Best Match Found:")
            print("Issue ID:", best.get("issue_id"))
            print("Score:", best.get("score"))
            print("Analysis:", best.get("analysis"))

        else:
            print("\n❌ No strong match found")

        print("\n" + "="*50)


if __name__ == "__main__":
    main()
