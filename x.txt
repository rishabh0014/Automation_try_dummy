import requests
from app.config import JIRA_BASE_URL, JIRA_USERNAME, JIRA_PASSWORD
from app.jira_utils import extract_issue_id


def get_issue_description(ims_link: str):
    issue_id = extract_issue_id(ims_link)

    if not issue_id:
        return None

    url = f"{JIRA_BASE_URL}/rest/api/2/issue/{issue_id}"

    response = requests.get(
        url,
        auth=(JIRA_USERNAME, JIRA_PASSWORD),
        headers={"Accept": "application/json"}
    )

    if response.status_code != 200:
        print(f"❌ Failed to fetch issue: {response.text}")
        return None

    data = response.json()

    # JIRA stores description here
    description = data.get("fields", {}).get("description")

    return description


def clean_description(desc):
    if not desc:
        return ""

    # Case 1: Plain string
    if isinstance(desc, str):
        return desc

    # Case 2: ADF format (JIRA JSON)
    if isinstance(desc, dict):
        return extract_text_from_adf(desc)

    return ""


def extract_text_from_adf(adf):
    texts = []

    def recurse(node):
        if isinstance(node, dict):
            if node.get("type") == "text":
                texts.append(node.get("text", ""))

            for child in node.get("content", []):
                recurse(child)

        elif isinstance(node, list):
            for item in node:
                recurse(item)

    recurse(adf)
    return " ".join(texts)

from app.jira_client import get_issue_description, clean_description

link = "https://ims.jira.samsungds.net/browser/PCIEJ-12106"

desc = get_issue_description(link)

cleaned = clean_description(desc)

print("RAW:\n", desc)
print("\nCLEANED:\n", cleaned)
