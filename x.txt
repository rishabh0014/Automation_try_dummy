from fastapi import FastAPI
from pydantic import BaseModel
from app.engine import process_issue

app = FastAPI()


class IssueRequest(BaseModel):
    testcase: str = None
    error_message: str = None
    ims_link: str = None


@app.get("/")
def health_check():
    return {"status": "running"}


@app.post("/analyze_issue")
def analyze_issue(issue: IssueRequest):
    issue_dict = issue.dict()

    result = process_issue(issue_dict)

    response = {
        "decision": result["decision"],
        "matched_issue": None,
        "score": None,
        "analysis": None
    }

    if result["best_match"]:
        response["matched_issue"] = result["best_match"].get("issue_id")
        response["score"] = result["best_match"].get("score")
        response["analysis"] = result["best_match"].get("analysis")

    return response
