script.js

function toggleFeedback() {

    const box = document.getElementById("feedback-box");

    box.classList.toggle("hidden");
}


async function submitFeedback() {

    const feedback =
        document.getElementById("feedback_text").value;

    if (!feedback.trim()) {
        alert("Please enter feedback");
        return;
    }

    try {

        const response = await fetch("/submit_feedback", {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify({
                feedback
            })
        });

        const data = await response.json();

        if (data.success) {

            document.getElementById(
                "feedback_status"
            ).innerText = "Feedback submitted successfully";

            document.getElementById(
                "feedback_text"
            ).value = "";

        } else {

            document.getElementById(
                "feedback_status"
            ).innerText = "Failed to submit feedback";
        }

    } catch (err) {

        console.error(err);

        document.getElementById(
            "feedback_status"
        ).innerText = "Error submitting feedback";
    }
}

main.py

class FeedbackRequest(BaseModel):
    feedback: str

import os
from datetime import datetime

@app.post("/submit_feedback")
async def submit_feedback(
    feedback_req: FeedbackRequest,
    request: Request
):

    try:

        os.makedirs("feedback", exist_ok=True)

        feedback_file = "feedback/feedback.txt"

        username = request.session.get("username", "unknown")

        client_ip = request.client.host

        timestamp = datetime.now().strftime(
            "%Y-%m-%d %H:%M:%S"
        )

        content = f"""
==================================================
Timestamp : {timestamp}
Username  : {username}
IP        : {client_ip}

Feedback:
--------------------------------------------------
{feedback_req.feedback}

==================================================

"""

        with open(feedback_file, "a", encoding="utf-8") as f:
            f.write(content)

        return {"success": True}

    except Exception as e:

        print(e)

        return {"success": False}
