    "HCore DTCM RW\n0x00061600")

    boxYellow(HCORE_DRAM_RO,
    "HCore DRAM RO\n0x01AE0000\n~599 KB code\n~105 KB margin")

    boxRed(CARVED,
    "CARVED FOR RM DRAM\n0x01B90000 - 0x01B9FFFF\n64 KB tail")

    boxBlue(HCORE_DRAM_RW,
    "HCore DRAM RW\n0x01BA0000 - 0x01BAFFFF\n64 KB")

    note right of HCORE_DRAM_RW
    This 64 KB tail was previously
    part of the HCore DRAM RO.

    It is now reused as:

    LOAD_RM_DRAM
    end note

    HCORE_ITCM -down-> HCORE_DTCM
    HCORE_DTCM -down-> HCORE_DRAM_RO
    HCORE_DRAM_RO -down-> CARVED
    CARVED -down-> HCORE_DRAM_RW
}

' ============================================================
' CROSS-PANEL FLOW
' ============================================================

HCORE_DRAM_RO -right-> RM_DRAM : 64 KB carved\nfrom HCore and reassigned\nto RM

RM_DRAM -right-> NEW_SECTION : Represented by new\nFW section index

NEW_SECTION ..> RM_DRAM : MF_CORE_DRAM_RO_SECIDX\n        ↓\nMF_RM_DRAM_RO_SECIDX

' ============================================================
' MPU relationship
' ============================================================

RM_DRAM ..down..> MPU21 : MPU Region 21\nprotects RM DRAM

' ============================================================
' INSERTION POINT
' ============================================================

note right of NEW_SECTION
Insertion point:

MF_HCORE_DRAM_RO_SECIDX
        ↓
MF_RM_DRAM_RO_SECIDX
        ↓
remaining sections

Therefore every section after
HCore DRAM RO moves by +1.
end note
