import os
import json

INPUT_FILE = "data/raw/issues.txt"
OUTPUT_FILE = "data/processed/issues.json"


def parse_issue_block(block):
    issue = {
        "issue_id": "",
        "testcase": "",
        "error_message": "",
        "analysis": "",
        "status": "",
        "ims_link": ""
    }

    lines = block.strip().split("\n")

    for line in lines:
        if ":" not in line:
            continue

        key, value = line.split(":", 1)
        key = key.strip()
        value = value.strip()

        if key in issue:
            issue[key] = value

    return issue


def parse_txt_file():
    with open(INPUT_FILE, "r", encoding="utf-8") as f:
        content = f.read()

    blocks = content.split("---ISSUE_START---")
    issues = []

    for block in blocks:
        if "---ISSUE_END---" not in block:
            continue

        block_content = block.split("---ISSUE_END---")[0].strip()

        if not block_content:
            continue

        try:
            issue = parse_issue_block(block_content)

            # Minimal validation
            if not issue["issue_id"]:
                continue

            issues.append(issue)

        except Exception as e:
            print(f"⚠️ Skipping malformed block: {e}")

    return issues


def main():
    print("📥 Parsing TXT file...")

    issues = parse_txt_file()

    print(f"✅ Parsed {len(issues)} issues")

    os.makedirs(os.path.dirname(OUTPUT_FILE), exist_ok=True)

    with open(OUTPUT_FILE, "w", encoding="utf-8") as f:
        json.dump(issues, f, indent=2)

    print(f"💾 Saved to {OUTPUT_FILE}")


if __name__ == "__main__":
    main()
